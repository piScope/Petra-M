cmake_minimum_required(VERSION 3.21)
project(SUPERLU_SOLVE LANGUAGES CXX)
find_package(MPI REQUIRED COMPONENTS CXX)
if(NOT NATIVE_PREFIX)
  message(FATAL_ERROR "NATIVE_PREFIX is required")
endif()
add_library(superlu_solve SHARED superlu_solve.cpp)
target_compile_features(superlu_solve PRIVATE cxx_std_17)
target_include_directories(superlu_solve PRIVATE "${NATIVE_PREFIX}/include")
foreach(component superlu_dist openblas metis GKlib)
  set(archive "${NATIVE_PREFIX}/lib/lib${component}.a")
  if(NOT EXISTS "${archive}")
    message(FATAL_ERROR "Required static PIC archive is missing: ${archive}")
  endif()
  target_link_libraries(superlu_solve PRIVATE "${archive}")
endforeach()
target_link_libraries(superlu_solve PRIVATE MPI::MPI_CXX ${CMAKE_DL_LIBS} m)
if(APPLE)
  target_link_options(superlu_solve PRIVATE
    "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/exported_symbols_list.txt")
  set_target_properties(superlu_solve PROPERTIES INSTALL_NAME_DIR "@rpath")
else()
  target_link_options(superlu_solve PRIVATE
    "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exported_version_script.txt"
    "LINKER:--no-undefined")
endif()
set_target_properties(superlu_solve PROPERTIES POSITION_INDEPENDENT_CODE ON)
install(TARGETS superlu_solve DESTINATION .)
